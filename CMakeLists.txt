cmake_minimum_required(VERSION 3.15)
project(TaskFolders VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Select Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
add_library(storage src/settings.cpp)
target_include_directories(storage PUBLIC src)
add_executable(TaskFolders WIN32 src/main.cpp src/editor.cpp src/overlay.cpp src/shell.cpp src/glass.cpp src/devices.cpp src/device_page.cpp src/media.cpp resources/editor.rc)
target_link_libraries(TaskFolders PRIVATE storage shell32 comdlg32 dwmapi ole32 advapi32 gdiplus comctl32)
target_link_options(TaskFolders PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/compatibility.manifest")
include(cmake/BatteryHub.cmake)
enable_testing()
add_executable(storage_tests tests/storage.cpp)
target_link_libraries(storage_tests PRIVATE storage)
add_test(NAME storage COMMAND storage_tests)
install(TARGETS TaskFolders RUNTIME DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
set(CPACK_GENERATOR ZIP)
include(CPack)
